\begin{frame}{Linear ODE Example\quad $2y'+4ty=8t$\quad $y(0)=3$}
\begin{itemize}
  \item Divide by 2 to get standard form
  \[
    y'+2ty=4t\qquad p(t)=2t\qquad q(t)=4t
  \]
  \item The integrating factor is
  \[
    \mu(t)=\exp\left(\int2t\,dt\right)=e^{t^2}
  \]
  \item Multiply and use the product rule, then integrate
  \[
    e^{t^2}y'+2te^{t^2}y=4te^{t^2}
    \quad\Longrightarrow\quad
    (e^{t^2}y)'=4te^{t^2}
  \]
  \[
    e^{t^2}y=\int4te^{t^2}\,dt+C=2e^{t^2}+C
  \]
  \item Use $y(0)=3$ to get $3=2+C$, so $C=1$. Divide by $e^{t^2}$ to get $\boxed{y(t)=2+e^{-t^2}}$
\end{itemize}
\end{frame}
