\begin{enumerate}
\item \begingroup
\setlength{\arraycolsep}{3pt}
Solve
$
  \vect{y}'=A\vect{y}+e^{-2t}\begin{pmatrix}-1\\-5\end{pmatrix}
  $, $\vect{y}(0)=\begin{pmatrix}2\\3\end{pmatrix}$
and
$A=\begin{pmatrix}1&-2\\2&1\end{pmatrix}$ with epairs
$
  1+2i,\, \begin{pmatrix}1\\-i\end{pmatrix};\, 1-2i,\, \begin{pmatrix}1\\i\end{pmatrix}
$
\endgroup
\begin{enumerate}
\item $\vect{y}_c(t)=\VectorBlank$
\item $\vect{y}_p(t)=\VectorBlank$
is a suitable UC form.
\item Row reducing
$\MatrixBlank$\hspace{0.12in} gives the UCs in
\par
$\vect{y}_p(t)=\VectorBlank$
\item  Row reducing
$\MatrixBlank$\hspace{0.12in} gives constants $\vect{c}$ in
\par
$\vect{y}(t)=\VectorBlank$
\end{enumerate}
