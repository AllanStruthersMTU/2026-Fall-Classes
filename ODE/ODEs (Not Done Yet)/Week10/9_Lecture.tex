  	\item Questions like HW with fewer boxes stuck together.

  \end{itemize}
\end{frame}
\begin{frame}{Separable and Linear ODEs}
\begin{block}{Separable\quad $y'=g(t)h(y)$}
\begin{itemize}
  \item Multiply by $dt$ and separate variables
  $\frac{dy}{h(y)}=g(t)\,dt$
  \item Integrate, then use the initial condition to determine $C$.
  \item Do not solve for $y$ on an exam. 
\end{itemize}
\end{block}
\begin{block}{Linear\quad $a_1(t)y'+a_0(t)y=f(t)\ \Longrightarrow\ y'+p(t)y=q(t)$}
\begin{itemize}
  \item Divide by $a_1(t)$ to get the 
  standard form $y'+p(t)y=q(t)$ 
  \item Multiply by the integrating factor
  $\mu(t)=\exp\left(\int p(t)\,dt\right)$ to get $(\mu y)'=\mu q(t)$.
  \item Integrate then use the initial condition to determine $C$.
\end{itemize}
\end{block}
\end{frame}
\begin{frame}{Linear Systems $\vect{y}'=\mat{A}\vect{y}+\vect{f}$: \quad $\vect{y}=\vect{y}_c+\vect{y}_p$
		for $\mat{A}\in\mathbb{R}^{n\times n}$}
\begin{block}{Complementary solution\quad $\vect{y}_c'=\mat{A}\vect{y}_c$}
\begin{itemize}
  \item Find $n$ linearly independent real solutions $\vect{y}_k(t)$ from eigenvalues and eigenvectors.
  \item Real eigenpairs give $e^{\lambda t}\vect{v}$. Splitting complex eigenpairs gives two LI solutions.
  Then $\vect{y}_c=\sum_{k=1}^{n}c_k\vect{y}_k$.
\end{itemize}
\end{block}
\begin{block}{Particular solution\quad $\vect{y}_p'=\mat{A}\vect{y}_p+\vect{f}(t)$}
\begin{itemize}
  \item Substitute the undetermined coefficient 
  form matching $\vect{f}$ and match coefficients.
  \item Add particular solutions for separate forcing terms.
\end{itemize}
\end{block}
For $\vect{y}(0)=\vect{y}_0$, determine $\vect{c}$ by solving
$\mat{M}\vect{c}=\vect{y}_0-\vect{y}_p(0)$,
where $\mat{M}=[\vect{y}_1(0)\ \cdots\ \vect{y}_n(0)]$.
\end{frame}

\begin{frame}{Variation of Parameters for $y'+p(t)y=q(t)$ with $y_1'=-p(t)y_1$
	and $y_c(t)=c y_1(t)$}
\begin{itemize}
  \item \keypoint{Variation of Parameters} replaces the constant $c$ by a function $u(t)$.
  \item Substitute $y_p(t)=u(t)y_1(t)$ and of course  
  $y_p'=u'y_1+uy_1'$ into the ODE
  \[
    y_p'+p(t)y_p
    =u'y_1+u\underbrace{\bigl(y_1'+p(t)y_1\bigr)}_{0}=q(t)
  \]
  So $ u'(t)y_1(t)=q(t)
    \quad
    \Longrightarrow
    u(t)=\int\frac{q(t)}{y_1(t)}\,dt + C
    $
 \item The general solution is
$y(t)=u(t)y_1(t)=\left(\int\frac{q(t)}{y_1(t)}\,dt+C\right)y_1(t)$.
\end{itemize}
\vfill
\[
  \boxed{y(t)=\left(C+\int\frac{q(t)}{y_1(t)}\,dt\right)y_1(t)}
\]
\end{frame}
\begin{frame}{Variation of Parameters for $\vect{y}'=\mat{A}\vect{y}+\vect{f}(t)$ with $\mat{\Phi}'=\mat{A}\mat{\Phi}$ and $\vect{y}_c=\mat{\Phi}\vect{c}$
	}
\begin{itemize}
  \item The \keypoint{Fundamental Matrix} has $n$ LI complementary solutions as columns
  \[
    \mat{\Phi}(t)=[\vect{y}_1(t)\ \cdots\ \vect{y}_n(t)]
  \]
  so $\mat{\Phi}'=[\vect{y}_1'\ \cdots\ \vect{y}_n']=[\mat{A}\vect{y}_1\ \cdots\ \mat{A}\vect{y}_n]=\mat{A}[\vect{y}_1\ \cdots\ \vect{y}_n]=\mat{A}\mat{\Phi}$
  \item Replace the constants in $\vect{y}_c=\mat{\Phi}\vect{c}$ by $\vect{u}(t)$.
  Substitute $\vect{y}=\mat{\Phi}\vect{u}$ into the ODE
  \[
    \begin{aligned}
      \mat{\Phi}'\vect{u}+\mat{\Phi}\vect{u}'
        &=\mat{A}\mat{\Phi}\vect{u}+\vect{f}\\
      \underbrace{(\mat{\Phi}'-\mat{A}\mat{\Phi})}_{0}\vect{u}
        +\mat{\Phi}\vect{u}'&=\vect{f}
    \end{aligned}
  \]
  \item So $\vect{u}'=\mat{\Phi}^{-1}\vect{f}$ and
  $\vect{u}(t)=\vect{c}+\int\mat{\Phi}(t)^{-1}\vect{f}(t)\,dt$.
\end{itemize}
\vfill
\[
  \boxed{\vect{y}(t)=\mat{\Phi}(t)\left(\vect{c}+\int\mat{\Phi}(t)^{-1}\vect{f}(t)\,dt\right)}
\]
\end{frame}
