\begin{enumerate}
\item \begingroup
\setlength{\arraycolsep}{3pt}
Solve
$
  \vect{y}'=A\vect{y}+e^{-2t}\begin{pmatrix}-7\\-7\end{pmatrix}
  $, $\vect{y}(0)=\begin{pmatrix}4\\0\end{pmatrix}$
and
$A=\begin{pmatrix}2&3\\3&2\end{pmatrix}$ with epairs
$
  5,\, \begin{pmatrix}1\\1\end{pmatrix};\, -1,\, \begin{pmatrix}1\\-1\end{pmatrix}
$
\endgroup
\begin{enumerate}
\item $\vect{y}_c(t)=\KeyBox{3.2in}{0.80in}{\begin{pmatrix}e^{5t}&e^{-t}\\e^{5t}&-e^{-t}\end{pmatrix}\vect{c}}\KeySide{2.5in}{\vect{y}_c=c_1e^{5t}\begin{pmatrix}1\\1\end{pmatrix}\\{}+c_2e^{-t}\begin{pmatrix}1\\-1\end{pmatrix}}$
\item $\vect{y}_p(t)=\KeyBox{3.2in}{0.80in}{e^{-2t}\vect{a}}$
is a suitable UC form.
\item Row reducing
$\KeyBox{1.8in}{0.80in}{\left[\begin{array}{c|c}-2I-A&\begin{matrix}-7\\-7\end{matrix}\end{array}\right]}$\hspace{0.12in} gives the UCs in
\par
$\vect{y}_p(t)=\KeyBox{3.2in}{0.80in}{e^{-2t}\begin{pmatrix}1\\1\end{pmatrix}}\KeySide{2.4in}{(-2I-A)\vect{a}=\begin{pmatrix}-7\\-7\end{pmatrix}\\-2I-A=\begin{pmatrix}-4&-3\\-3&-4\end{pmatrix}\\\vect{a}=\begin{pmatrix}1\\1\end{pmatrix}}$
\item  Row reducing
$\KeyBox{1.8in}{0.80in}{\left[\begin{array}{rr|r}1&1&3\\1&-1&-1\end{array}\right]}\KeySide{2.8in}{\left[\begin{pmatrix}1&1\\1&-1\end{pmatrix}\mid\vect{y}(0)-\vect{a}\right]\\\begin{pmatrix}4\\0\end{pmatrix}-\vect{a}=\begin{pmatrix}3\\-1\end{pmatrix}\\\vect{c}=\begin{pmatrix}1\\2\end{pmatrix}}$
\par
gives constants $\vect{c}$ in
\par
$\vect{y}(t)=\KeyBox{3.2in}{0.80in}{\begin{pmatrix}e^{5t}+2e^{-t}\\e^{5t}-2e^{-t}\end{pmatrix}+e^{-2t}\begin{pmatrix}1\\1\end{pmatrix}}$
\end{enumerate}
\newpage
\item \begingroup
\setlength{\arraycolsep}{3pt}
Solve
$
  \vect{y}'=A\vect{y}+\cos(2t)\begin{pmatrix}5\\5\end{pmatrix}
  $, $\vect{y}(0)=\begin{pmatrix}0\\-2\end{pmatrix}$
and
$A=\begin{pmatrix}2&-1\\3&-2\end{pmatrix}$ with epairs
$
  1,\, \begin{pmatrix}1\\1\end{pmatrix};\, -1,\, \begin{pmatrix}1\\3\end{pmatrix}
$
\endgroup
\begin{enumerate}
\item $\vect{y}_c(t)=\KeyBox{3.2in}{0.80in}{\begin{pmatrix}e^t&e^{-t}\\e^t&3e^{-t}\end{pmatrix}\vect{c}}\KeySide{2.5in}{\vect{y}_c=c_1e^t\begin{pmatrix}1\\1\end{pmatrix}\\{}+c_2e^{-t}\begin{pmatrix}1\\3\end{pmatrix}}$
\item $\vect{y}_p(t)=\KeyBox{3.2in}{0.80in}{\vect{a}\cos(2t)+\vect{b}\sin(2t)}$
is a suitable UC form.
\item Row reducing
$\KeyBox{3.0in}{1.60in}{\left[\begin{array}{cc|c}-A&2I&\begin{matrix}5\\5\end{matrix}\\-2I&-A&\begin{matrix}0\\0\end{matrix}\end{array}\right]}$\hspace{0.12in} gives the UCs in
\par
$\vect{y}_p(t)=\KeyBox{3.2in}{0.80in}{\begin{pmatrix}-1\\-1\end{pmatrix}\cos(2t)+\begin{pmatrix}2\\2\end{pmatrix}\sin(2t)}\KeySide{2.4in}{\begin{pmatrix}-A&2I\\-2I&-A\end{pmatrix}\begin{pmatrix}\vect{a}\\\vect{b}\end{pmatrix}=\begin{pmatrix}5\\5\\0\\0\end{pmatrix}\\=\begin{pmatrix}-2&1&2&0\\-3&2&0&2\\-2&0&-2&1\\0&-2&-3&2\end{pmatrix}\begin{pmatrix}\vect{a}\\\vect{b}\end{pmatrix}\\\vect{a}=\begin{pmatrix}-1\\-1\end{pmatrix},\quad\vect{b}=\begin{pmatrix}2\\2\end{pmatrix}}$
\item Row reducing
$\KeyBox{1.8in}{0.80in}{\left[\begin{array}{rr|r}1&1&1\\1&3&-1\end{array}\right]}\KeySide{2.8in}{\left[\begin{pmatrix}1&1\\1&3\end{pmatrix}\mid\vect{y}(0)-\vect{a}\right]\\\begin{pmatrix}0\\-2\end{pmatrix}-\vect{a}=\begin{pmatrix}1\\-1\end{pmatrix}\\\vect{c}=\begin{pmatrix}2\\-1\end{pmatrix}}$
\par
gives constants $\vect{c}$ in
\par
$\vect{y}(t)=\KeyBox{3.2in}{0.80in}{\begin{pmatrix}2e^t-e^{-t}\\2e^t-3e^{-t}\end{pmatrix}+\vect{y}_p(t)}$
\end{enumerate}
\end{enumerate}
