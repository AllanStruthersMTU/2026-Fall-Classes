\begin{frame}{ODE Definitions and Terminology}
\begin{itemize}
  \item An \keypoint{IVP} is an \keypoint{ODE} relating $\vect{y}(t)$ to $\vect{y}'(t),\vect{y}''(t),\ldots$ with an \keypoint{IC} $\vect{y}(t_0)=\vect{y}_0$
  \begin{itemize}
    \item \keypoint{Order} is the highest derivative order. \item A \keypoint{Solution} satisfies ODE and IC for $a<t_0<b$.
    \item No explicit $t$ dependence is \keypoint{Autonomous}. An \keypoint{Equilibrium} 
    $f(y^*)=0$ gives a constant solution $\vect{y}(t)=\vect{y}_*$.
  \end{itemize}
  \item \keypoint{Linear} means $\vect{y}'=\mat{A}(t)\vect{y}+\vect{b}(t)$, with coefficients independent of $\vect{y}$
  \begin{itemize}
    \item \keypoint{Constant Coefficient (CC)} means $\mat{A}$ and $\vect{b}$ are independent of $t$
    \item $\vect{b}(t)$ is the \keypoint{Forcing}. Zero forcing means \keypoint{Homogeneous}
  \end{itemize}
  \item First order scalar ODEs
  \begin{itemize}
    \item \keypoint{Separable} means $y'=g(t)h(y)$
    \item \keypoint{Standard Form} for 1st order linear is  $y'+p(t)y=q(t)$.
    \end{itemize}
    \item Modern numerical solvers give reliable numerical values for IVPs.  
\end{itemize}
\end{frame}
