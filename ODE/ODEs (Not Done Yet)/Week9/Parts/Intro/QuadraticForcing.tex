\begin{frame}{Quadratic Forcing \qquad $\vect{y}'=\overbrace{\begin{pmatrix}-2&1\\1&-2\end{pmatrix}}^{\mat{A}}\vect{y}+\overbrace{\begin{pmatrix}6\\-3\end{pmatrix}t^2+\begin{pmatrix}0\\3\end{pmatrix}t+\begin{pmatrix}0\\3\end{pmatrix}}^{\vect{f}(t)}$}
\small
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{itemize}
\setlength{\itemsep}{3pt}
  \item Try $\vect{y}_p(t)=\vect{a}t^2+\vect{b}t+\vect{d}$ which gives
  $\vect{y}_p'=2\vect{a}t+\vect{b}$ in the ODE and match powers of $t$
  \[
    {\color{blue}2\vect{a}t}+{\color{green!50!black}\vect{b}}
    ={\color{red}\mat{A}\vect{a}t^2}
    +{\color{blue}\mat{A}\vect{b}t}
    +{\color{green!50!black}\mat{A}\vect{d}}
    +{\color{red}\begin{pmatrix}6\\-3\end{pmatrix}t^2}
    +{\color{blue}\begin{pmatrix}0\\3\end{pmatrix}t}
    +{\color{green!50!black}\begin{pmatrix}0\\3\end{pmatrix}}
  \]
  \[
    \color{red}
    \mat{A}\vect{a}=-\begin{pmatrix}6\\-3\end{pmatrix}
    \quad\Longrightarrow\quad
    \left[\begin{array}{rr|r}-2&1&-6\\1&-2&3\end{array}\right]
    \sim\left[\begin{array}{rr|r}1&0&3\\0&1&0\end{array}\right]
    \quad\Longrightarrow\quad\vect{a}=\begin{pmatrix}3\\0\end{pmatrix}
  \]
  \[
    \color{blue}
    \mat{A}\vect{b}=2\vect{a}-\begin{pmatrix}0\\3\end{pmatrix}
    =\begin{pmatrix}6\\-3\end{pmatrix}
    \quad\Longrightarrow\quad
    \left[\begin{array}{rr|r}-2&1&6\\1&-2&-3\end{array}\right]
    \sim\left[\begin{array}{rr|r}1&0&-3\\0&1&0\end{array}\right]
    \quad\Longrightarrow\quad\vect{b}=\begin{pmatrix}-3\\0\end{pmatrix}
  \]
  \[
    \color{green!50!black}
    \mat{A}\vect{d}=\vect{b}-\begin{pmatrix}0\\3\end{pmatrix}
    =\begin{pmatrix}-3\\-3\end{pmatrix}
    \quad\Longrightarrow\quad
    \left[\begin{array}{rr|r}-2&1&-3\\1&-2&-3\end{array}\right]
    \sim\left[\begin{array}{rr|r}1&0&3\\0&1&3\end{array}\right]
    \quad\Longrightarrow\quad\vect{d}=\begin{pmatrix}3\\3\end{pmatrix}
  \]
  \item $\displaystyle \vect{y}(t)=\begin{pmatrix}3t^2-3t+3\\3\end{pmatrix} +c_1e^{-t}\begin{pmatrix}1\\1\end{pmatrix} +c_2e^{-3t}\begin{pmatrix}1\\-1\end{pmatrix} \sim\begin{pmatrix}3t^2\\3\end{pmatrix}\text{ as }t\to\infty$
\end{itemize}
\end{frame}
