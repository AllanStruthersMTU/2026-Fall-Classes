\begin{frame}{Complex Exponentials
	\quad
$e^{(\alpha \pm \beta i t)} = e^{\alpha t} e^{\pm \beta it} $}
Maclaurin/Taylor series 
\[
    \setlength{\arraycolsep}{2pt}
    \begin{array}{rcrcr@{}lcr@{}lcr@{}lcr@{}lcl}
      e^x & = & 1 & + & 1 & x & + & (1/2!) & x^2 & + & (1/3!) & x^3 & + & (1/4!) & x^4 & + & \cdots \\[4pt]
      \cos(x) & = & 1 & + & 0 & x & - & (1/2!) & x^2 & + & 0 & x^3 & + & (1/4!) & x^4 & + & \cdots \\[4pt]
      \sin(x) & = & 0 & + & 1 & x & + & 0 & x^2 & - & (1/3!) & x^3 & + & 0 & x^4 & + & \cdots
    \end{array}
\]
\[
    \setlength{\arraycolsep}{2pt}
    \begin{array}{rcrcr@{}lcr@{}lcr@{}lcr@{}lcl}
      e^{iy} & = & {\color{blue}1} & {\color{red}+} & {\color{red}i} & {\color{red}y} & {\color{blue}-} & {\color{blue}(1/2!)} & {\color{blue}y^2} & {\color{red}-} & {\color{red}(i/3!)} & {\color{red}y^3} & {\color{blue}+} & {\color{blue}(1/4!)} & {\color{blue}y^4} & + & \cdots \\[4pt]
      {\color{blue}\cos(y)} & = & {\color{blue}1} & {\color{blue}+} & {\color{blue}0} & {\color{blue}y} & {\color{blue}-} & {\color{blue}(1/2!)} & {\color{blue}y^2} & {\color{blue}+} & {\color{blue}0} & {\color{blue}y^3} & {\color{blue}+} & {\color{blue}(1/4!)} & {\color{blue}y^4} & + & \cdots \\[4pt]
      {\color{red}i\sin(y)} & {\color{red}=} & {\color{red}0} & {\color{red}+} & {\color{red}i} & {\color{red}y} & {\color{red}+} & {\color{red}0} & {\color{red}y^2} & {\color{red}-} & {\color{red}(i/3!)} & {\color{red}y^3} & {\color{red}+} & {\color{red}0} & {\color{red}y^4} & {\color{red}+} & {\color{red}\cdots} \\[4pt]
      {\color{blue}\cos(y)}{\color{red}+i\sin(y)} & = & {\color{blue}1} & {\color{red}+} & {\color{red}i} & {\color{red}y} & {\color{blue}-} & {\color{blue}(1/2!)} & {\color{blue}y^2} & {\color{red}-} & {\color{red}(i/3!)} & {\color{red}y^3} & {\color{blue}+} & {\color{blue}(1/4!)} & {\color{blue}y^4} & + & \cdots
    \end{array}
\]
So $\boxed{e^{i y} = {\color{blue}\cos(y)}+{\color{red}i\sin(y)}}$ and
$
  \boxed{e^{(\alpha\pm\beta i)t}={\color{blue}e^{\alpha t}}({\color{blue}\cos(\beta t)}\pm {\color{red}i\sin(\beta t)})}
$
\end{frame}
