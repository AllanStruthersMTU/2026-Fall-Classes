\begin{frame}{General solution 
		$\vect{y}(t)=\vect{y}_c(t)+\vect{y}_p(t)$ of
		$\vect{y}'(t)=\mat{A}\vect{y}(t) + \vect{f}(t)$}
\setlength{\abovedisplayskip}{4pt}
\setlength{\belowdisplayskip}{4pt}
\setlength{\abovedisplayshortskip}{2pt}
\setlength{\belowdisplayshortskip}{2pt}
\begin{itemize}
\setlength{\itemsep}{2pt}
  \item The \keypoint{Complementary Solution} satisfies
  $\vect{y}_c'=\mat{A}\vect{y}_c$ with an eigen-basis
  $\mat{A}\vect{v}_i=\lambda_i\vect{v}_i$
  \[
  \vect{y}_c(t)=\sum_{i=1}^n c_i e^{\lambda_i t}\vect{v}_i=\underbrace{\begin{pmatrix}\vect{v}_1&\cdots&\vect{v}_n\end{pmatrix}}_{V}\begin{pmatrix}e^{\lambda_1 t}& &0\\ &\ddots& \\0& &e^{\lambda_n t}\end{pmatrix}\underbrace{\begin{pmatrix}c_1\\\vdots\\c_n\end{pmatrix}}_{\vect{c}}
  \]
  \item 
  A \keypoint{Particular Solution} satisfies
  $\vect{y}_p'=\mat{A}\vect{y}_p+\vect{f}(t)$
  with determined coefficients.
  \begin{itemize}
  \setlength{\itemsep}{1pt}
    \item If $\vect{f}(t)$ is polynomial guess
    $\vect{y}_p(t)=\vect{a}_0+\vect{a}_1t+\dots$
    \item If $\vect{f}(t)$ is exponential guess $\vect{y}_p(t)=\vect{a}e^{\alpha t}$ with matching $\alpha$. 
    \item If $\vect{f}(t)$ is $\cos$ and/or $\sin $ guess
    $\vect{y}_p(t)=\vect{a}\cos(\omega t)+\vect{b}\sin(\omega t)$ with matching $\omega$
    \item Undetermined coefficients $\vect{a}_i$ etc are determined by plugging into $\vect{y}_p'=\mat{A}\vect{y}_p+\vect{f}(t)$
  \end{itemize}
  \item Solving $
  V\vect{c}=\vect{y}_0-\vect{y}_p(0)$  gives 
  $\vect{c}$ satisfying the \keypoint{IC} 
  $\vect{y}(0) = \vect{y}_c(0)+\vect{y}_p(0)=\vect{y}_0$.
\end{itemize}
\end{frame}
