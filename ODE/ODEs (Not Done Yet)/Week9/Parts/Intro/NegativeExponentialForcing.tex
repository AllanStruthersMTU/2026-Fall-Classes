\begin{frame}{Exponential Forcing \qquad $\vect{y}'=\overbrace{\begin{pmatrix}-2&1\\1&-2\end{pmatrix}}^{\mat{A}}\vect{y}+\overbrace{\begin{pmatrix}7\\2\end{pmatrix}e^{-4t}}^{\vect{f}(t)}$}
\small
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{itemize}
\setlength{\itemsep}{3pt}
  \item Try $\vect{y}_p(t)=\vect{a}e^{-4t}$ which gives
  $\vect{y}_p'=-4\vect{a}e^{-4t}$ in the ODE
  \[
    -4\vect{a}{\color{red}e^{-4t}}=\mat{A}\vect{a}{\color{red}e^{-4t}}
    +\begin{pmatrix}7\\2\end{pmatrix}{\color{red}e^{-4t}}
    \quad\Longrightarrow\quad
    (-4\mat{I}-\mat{A})\vect{a}=\begin{pmatrix}7\\2\end{pmatrix}
  \]
  \[
    \left(\begin{array}{rr}-2&-1\\-1&-2\end{array}\right)\begin{pmatrix}a_1\\a_2\end{pmatrix}
    =\begin{pmatrix}7\\2\end{pmatrix}
    \quad\Longrightarrow\quad
    \left[\begin{array}{rr|r}-2&-1&7\\-1&-2&2\end{array}\right]
    \sim\left[\begin{array}{rr|r}1&2&-2\\0&3&3\end{array}\right]
    \sim\left[\begin{array}{rr|r}1&0&-4\\0&1&1\end{array}\right]
  \]
  \item Thus $\vect{a}=\begin{pmatrix}a_1\\a_2\end{pmatrix}=\begin{pmatrix}-4\\1\end{pmatrix}$
  and $\vect{y}_p(t)=\begin{pmatrix}-4\\1\end{pmatrix}e^{-4t}$.
  \item General solution
  \[
    \vect{y}(t)=\vect{y}_p(t)+\vect{y}_c(t)
    =\begin{pmatrix}-4\\1\end{pmatrix}e^{-4t}
    +c_1e^{-t}\begin{pmatrix}1\\1\end{pmatrix}
    +c_2e^{-3t}\begin{pmatrix}1\\-1\end{pmatrix}
  \]
  \item As always, initial conditions $\vect{y}(0)=\vect{y}_0$ determine the constants.
\end{itemize}
\end{frame}
