\begin{frame}{First Order Homogeneous Linear System\quad $\vect{y}'=\mat{A}\vect{y}$}
\begin{itemize}
  \item For constant $\mat{A}\in\mathbb{R}^{n\times n}$, compute eigenpairs
  \[
    \mat{A}\vect{v}_i=\lambda_i\vect{v}_i
    \qquad \vect{v}_i\ne\vect{0}
  \]
  \item Each eigenpair gives a solution
  \[
    \vect{y}_{c,i}(t)=e^{\lambda_i t}\vect{v}_i
    \qquad
    \vect{y}_{c,i}'(t)=\lambda_i e^{\lambda_i t}\vect{v}_i
    =\mat{A}\vect{y}_{c,i}(t)
  \]
  \item If $\vect{v}_1,\ldots,\vect{v}_n$ are linearly independent, the \keypoint{General Solution (GS)} is
  \[
    \boxed{\vect{y}_c(t)=c_1e^{\lambda_1t}\vect{v}_1+\cdots+c_ne^{\lambda_nt}\vect{v}_n}
  \]
  \item The IC $\vect{y}(0)=\vect{y}_0$ determines $\vect{c}=(c_1,\ldots,c_n)$ through the LS
  \[
    \mat{V}\vect{c}=\vect{y}_0
    \qquad \mbox{with} \quad  \mat{V}=(\vect{v}_1\ \cdots\ \vect{v}_n)
  \]
\end{itemize}
\end{frame}
