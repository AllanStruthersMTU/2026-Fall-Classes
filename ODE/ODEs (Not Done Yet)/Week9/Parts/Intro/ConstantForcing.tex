\begin{frame}{Constant Forcing \qquad $\vect{y}'=\overbrace{\begin{pmatrix}-2&1\\1&-2\end{pmatrix}}^{\mat{A}}\vect{y}+\overbrace{\begin{pmatrix}3\\0\end{pmatrix}}^{\vect{f}}$}
\small
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{itemize}
\setlength{\itemsep}{3pt}
  \item Try 
  $\vect{y}_p(t)=\vect{a} =
  \begin{pmatrix}a_1\\a_2\end{pmatrix}$ which gives 
  $\vect{y}_p'=\vect{0}$ in the ODE
  \[
    \vect{0}=\mat{A}\begin{pmatrix}a_1\\a_2\end{pmatrix}+\vect{f}
    \quad\Longrightarrow\quad
    \mat{A}\begin{pmatrix}a_1\\a_2\end{pmatrix}=-\vect{f}
    \quad\Longrightarrow\quad
    \begin{pmatrix}-2&1\\1&-2\end{pmatrix}
    \begin{pmatrix}a_1\\a_2\end{pmatrix}
    =\begin{pmatrix}-3\\0\end{pmatrix}.
  \]
  \[
    \left[\begin{array}{rr|r}-2&1&-3\\1&-2&0\end{array}\right]
    \ \sim\ 
    \left[\begin{array}{rr|r}1&-2&0\\0&-3&-3\end{array}\right]
    \ \sim\ 
    \left[\begin{array}{rr|r}1&0&2\\0&1&1\end{array}\right]
    \quad\Longrightarrow\quad
    \vect{y}_p(t)=\begin{pmatrix}a_1\\a_2\end{pmatrix}
    =\begin{pmatrix}2\\1\end{pmatrix}.
  \]
  \item General solution
  \[
    \vect{y}(t)=\vect{y}_p(t)+\vect{y}_c(t)
    =\begin{pmatrix}2\\1\end{pmatrix}
    +c_1e^{-t}\begin{pmatrix}1\\1\end{pmatrix}
    +c_2e^{-3t}\begin{pmatrix}1\\-1\end{pmatrix}.
  \]
  \item An IC $\vect{y}(0)=\vect{y}_0$ determine the 
  constants $\vect{c}$ from the linear system
  \[
    \left(\begin{array}{rr}1&1\\1&-1\end{array}\right)
    \begin{pmatrix}c_1\\c_2\end{pmatrix}
    =\vect{y}_0-\begin{pmatrix}2\\1\end{pmatrix}
    .
  \]
\end{itemize}
\end{frame}
