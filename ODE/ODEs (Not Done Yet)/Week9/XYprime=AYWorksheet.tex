\newcommand{\EigenvalueSystemProblem}[3]{%
  \begingroup
  \setlength{\parskip}{3pt}
  \textbf{Problem #1}\quad Find the GS, then use the IC to determine the constants
  \[
    \vect{y}'=A\vect{y}
    \qquad A=\begin{pmatrix}#2\end{pmatrix}
    \qquad \vect{y}(0)=\begin{pmatrix}#3\end{pmatrix}
  \]
  \textbf{#1a. Eigenvalues and eigenvectors}\par
  $\displaystyle\det(A-\lambda I)=\SystemBlank{3.5in}{0.7in}=0$\par
  \vspace{0.04in}
  $\displaystyle\lambda_1=\SystemBlank{0.55in}{0.55in}\qquad
    \vect{v}_1=\SystemVectorBlank{1in}{1in}$\par
  \vspace{0.04in}
  $\displaystyle\lambda_2=\SystemBlank{0.55in}{0.55in}\qquad
    \vect{v}_2=\SystemVectorBlank{1in}{1in}$\par
  \vfill
  \textbf{#1b. General solution}\par
  $\displaystyle\vect{y}_c(t)=\SystemBlank{0.46\linewidth}{1in}$\par
  Write down $B\vect{c}=\vect{b}$ to solve for $\vect{c}=\begin{pmatrix}c_1\\c_2\end{pmatrix}$\par
  $\displaystyle B=\SystemVectorBlank{1.25in}{1.25in}\qquad
    \vect{b}=\SystemVectorBlank{0.65in}{1in}$\par
  $\displaystyle\vect{y}(t)=\SystemBlank{0.46\linewidth}{1in}$\par
  \endgroup
}
