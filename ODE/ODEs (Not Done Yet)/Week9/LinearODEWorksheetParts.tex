% Front problem. The IC selects t>0.
% Instructor check: p=ln(t). Integration by parts gives integral p=t*ln(t)-t.
% Choose mu=exp(t*ln(t)-t). Then mu*q=1, so mu*y=t+C.
% The IC y(1)=0 gives C=-1. Thus y=(t-1)exp(t-t*ln(t)).
\LinearProblem{1}{2y'+2\ln(t)y=2\exp\left(t-t\ln(t)\right)}{1}{0}
\newpage
% Back problem.
% Instructor check: p=t*exp(t), integral p=(t-1)exp(t) by parts.
% Choose mu=exp((t-1)exp(t)). Then mu*q=2t, so mu*y=t^2+C.
% mu(1)=1 and y(1)=2 give C=1. The solution exists for all real t.
\LinearProblem{2}{3y'+3te^t y=6t\exp\left((1-t)e^t\right)}{1}{2}
