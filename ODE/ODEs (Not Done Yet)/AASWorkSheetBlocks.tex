% Answer boxes use {rows}{columns}{point size}{phantom text}.
% Augmented boxes require at least two columns.
% Build one row, then repeat that row. The caller supplies \AnswerCell.
\newcommand{\AnswerArray}[2]{
  \gdef\AnswerRow{\AnswerCell}
  \foreach \col in {1,...,#2-1} {
    % Expand row with seperators and cells
    \xappto\AnswerRow{& \AnswerCell}
  }
  \gdef\AnswerBody{}
  \foreach \row in {1,...,#1} {
    \gappto\AnswerBody{\AnswerRow\\}
  }
  \begin{array}{*{#2}{c}}\AnswerBody\end{array}
}
