\begin{enumerate}
\item \textbf{Two-Story Building Practice}
\par\noindent\resizebox{0.70\linewidth}{!}{\TwoStoryBuilding{%
  basemotion={2\sin(t)},mass1=3,mass2=2,
  stiffness1=6,stiffness2=4,damping1=3,damping2=2}}
\par Displacements are in m and time is in s. The matrices have units
kg, N s/m, and N/m, respectively, and $\vect{b}$ is in N.
\begin{enumerate}
\item For $\mat{M}\vect{y}''+\mat{C}\vect{y}'+\mat{K}\vect{y}=\vect{b}(t)$
\[
\boxed{\mat{M}=\begin{pmatrix}3&0\\0&2\end{pmatrix}}
\qquad
\boxed{\mat{C}=\begin{pmatrix}5&-2\\-2&2\end{pmatrix}}
\qquad
\boxed{\mat{K}=\begin{pmatrix}10&-4\\-4&4\end{pmatrix}}
\]
\[
\boxed{\vect{b}(t)=\begin{pmatrix}6y_0(t)+3y_0'(t)\\0\end{pmatrix}}
\]
The base spring and damper act on the first floor. With
$y_0(t)=2\sin(t)$ this becomes
\[
\vect{b}(t)=\begin{pmatrix}12\sin(t)+6\cos(t)\\0\end{pmatrix}
\]
\item With $\vect{z}=(y_1,y_2,y_1',y_2')$
\[
\boxed{\mat{A}=\begin{pmatrix}
\mat{0}&\mat{I}\\
-\mat{M}^{-1}\mat{K}&-\mat{M}^{-1}\mat{C}
\end{pmatrix}}
\]
For completeness, the forcing in $\vect{z}'=\mat{A}\vect{z}+\vect{g}(t)$ is
\[
\vect{g}(t)=\begin{pmatrix}0\\0\\2y_0(t)+y_0'(t)\\0\end{pmatrix}
=\begin{pmatrix}0\\0\\4\sin(t)+2\cos(t)\\0\end{pmatrix}
\]
\end{enumerate}
\newpage
\item \textbf{Linear Initial Value Problem}
\[
y'-\frac{4}{t}y=2t^5
\qquad y(1)=3
\]
\begin{enumerate}
\item The integrating factor is
\[
\mu(t)=\exp\left(\int\boxed{-4/t}\,dt\right)=\boxed{t^{-4}}
\]
since $\int(-4/t)\,dt=-4\ln|t|$.
\vfill
\item Multiplying by $\mu(t)$ gives
\[
(t^{-4}y)'=\boxed{2t}
\]
\[
t^{-4}y=\int\boxed{2t}\,dt+C=\boxed{t^2}+C
\]
\vfill
\item The initial condition gives
$3=1+C$, so $C=2$ and
\[
\boxed{t^{-4}y=t^2+2}
\]
Check
\[
(t^{-4}y)'=2t
\qquad 1^{-4}y(1)=1^2+2=3
\]
\end{enumerate}
\vfill
\end{enumerate}
