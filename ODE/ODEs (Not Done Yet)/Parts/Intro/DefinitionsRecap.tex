\begin{frame}{ODE Definitions and Terminology}
\begin{itemize}
  \item An \keypoint{ODE} relates functions of one variable to their derivatives
  \begin{itemize}
  \item \keypoint{Order} is the highest derivative order present
  \item \keypoint{Autonomous} means no explicit dependence on $t$
  \item An \keypoint{Initial Condition (IC)} specifies $\vect{y}(t_0)=\vect{y}_0$
  \item An \keypoint{Initial Value Problem (IVP)} is an \keypoint{ODE} with initial conditions
  \item A \keypoint{Solution} satisfies the \keypoint{ODE} and IC on an interval containing $t_0$
  \end{itemize}
  \item $\vect{y}'=\mat{A}(t)\vect{y}+\vect{b}(t)$
  is a \keypoint{Linear ODE} if the  coefficients are independent of $\vect{y}$.
  \begin{itemize}
  	\item \keypoint{Constant Coefficient (CC)} means $\mat{A}$ and $\vect{b}$ do not depend on $t$
  \item $\vect{b}(t)$ is the \keypoint{Forcing}. If
  $\vect{b}(t)=\vect{0}$ the \keypoint{ODE} is \keypoint{Homogeneous}.  
    \end{itemize}
\end{itemize}
\vspace{4pt}\hrule\vspace{4pt}
\begin{itemize}
  \item An \keypoint{Equilibrium Solution} of $\vect{y}'=\vect{f}(\vect{y})$ is constant $\vect{y}(t)=\vect{y}_*$ with $\vect{f}(\vect{y}_*)=\vect{0}$
\end{itemize}
\end{frame}
