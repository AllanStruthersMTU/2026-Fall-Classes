\begin{frame}{Euler's Method}
\begin{itemize}
  \item For the IVP $\vect{y}'=\vect{f}(t,\vect{y})$,
  $\vect{y}(t_0)=\vect{y}_0$, choose step $h>0$
  \[
    \vect{y}'(t_n)\approx\frac{\vect{y}_{n+1}-\vect{y}_n}{h}
    \quad\Longrightarrow\quad
    \vect{y}_{n+1}=\vect{y}_n+h\vect{f}(t_n,\vect{y}_n)
    \qquad t_{n+1}=t_n+h
  \]
  \item For the autonomous R\"ossler system $\vect{y}'=\vect{f}(\vect{y})$ with $\vect{y}(0)=(1,0,0)$ and $h=0.1$
  \[
    \vect{y}_1=\vect{y}_0+0.1\vect{f}(\vect{y}_0)
    =(1,0,0)+0.1(0,1,0.2)=(1.0,0.1,0.02)
  \]
  \[
    \vect{f}(\vect{y})=
    \begin{pmatrix}
      -y_2-y_3\\
      y_1+0.2y_2\\
      0.2+y_3(y_1-5.7)
    \end{pmatrix}
    \qquad
    \begin{array}{ccl}
      n&t_n&\multicolumn{1}{c}{\vect{y}_n}\\ \hline
      0&0.0&(1.0000,\ 0.0000,\ 0.0000)\\
      1&0.1&(1.0000,\ 0.1000,\ 0.0200)\\
      2&0.2&(0.9880,\ 0.2020,\ 0.0306)\\
      3&0.3&(0.9647,\ 0.3048,\ 0.0362)
    \end{array}
  \]
  \item Practical solvers use more sophisticated methods with error control. Small steps where needed, big steps where possible! 
\end{itemize}
\end{frame}
