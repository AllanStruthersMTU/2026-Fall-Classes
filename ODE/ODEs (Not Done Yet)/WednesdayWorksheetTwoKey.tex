\begin{enumerate}
\item Consider $\vect{y}'=A\vect{y}$ with
$A=\begin{pmatrix}3&1&0\\-1&5&0\\-1&8&-3\end{pmatrix}\quad \vect{y}(0)=\vect{y}_0=\begin{pmatrix}5\\8\\2\end{pmatrix}$\newline
Eigenvalues $4,4,-3$. \quad One LI eigenvector for $4$ is $\vect{v}_1=\begin{pmatrix}1\\1\\1\end{pmatrix}$. \quad For $-3$, $\vect{v}_3=\begin{pmatrix}0\\0\\1\end{pmatrix}$.
\begin{enumerate}
\item Use $\vect{w}$ for the extra vector. A form for the 2\textsuperscript{ND} solution associated with $\lambda=4$ is
\par $\vect{y}_2(t)=\keybox{5.65in}{1.05in}{e^{4t}(t\vect{v}_1+\vect{w})}$
\vfill
\item The augmented matrix you would row reduce to find $\vect{w}$ is
\par $M_w=\keybox{5.65in}{1.05in}{\left[\begin{array}{c|c}A-4I&\vect{v}_1\end{array}\right]
\quad\text{or}\quad
\left[\begin{array}{rrr|r}-1&1&0&1\\-1&1&0&1\\-1&8&-7&1\end{array}\right]}$
\par{\footnotesize $M_w\ \overset{R_3\gets R_3-R_1}{\sim}\
\left[\begin{array}{rrr|r}
-1&1&0&1\\-1&1&0&1\\0&7&-7&0
\end{array}\right]$
\qquad One choice is $\vect{w}=\begin{pmatrix}0\\1\\1\end{pmatrix}$.}
\vfill
\item The augmented matrix you would row reduce to satisfy the IC is
\par $M_{\mathrm{IC}}=\keybox{5.65in}{1.05in}{\left[\begin{array}{ccc|c}\vect{v}_1&\vect{w}&\vect{v}_3&\vect{y}_0\end{array}\right]
\quad\text{or}\quad\left[\begin{array}{rrr|r}1&0&0&5\\1&1&0&8\\1&1&1&2\end{array}\right]}$
\vfill
\item Not on the exam
\par $\vect{y}(t)=\keybox{5.65in}{1.25in}{e^{4t}\bigl((5+3t)\vect{v}_1+3\vect{w}\bigr)-6e^{-3t}\vect{v}_3}$
\end{enumerate}

\newpage
\item Consider $\vect{y}'=A\vect{y}$ with
$A=\begin{pmatrix}-7&4&0\\-8&1&0\\-8&-4&5\end{pmatrix}\quad \vect{y}(0)=\begin{pmatrix}6\\2\\9\end{pmatrix}$\newline
Eigenvalues $-3\pm4i,\,5$. \quad For $-3+4i$, $\vect{v}=\begin{pmatrix}1\\1+i\\1+i\end{pmatrix}$. \quad For $5$, $\vect{v}_3=\begin{pmatrix}0\\0\\1\end{pmatrix}$.
\begin{enumerate}
\item A real complementary solution is
\par $\vect{y}_c(t)=\keybox{5.65in}{1.45in}{\begin{gathered}e^{-3t}\begin{pmatrix}c&s\\c-s&c+s\\c-s&c+s\end{pmatrix}\begin{pmatrix}c_1\\c_2\end{pmatrix}+c_3e^{5t}\vect{v}_3\\c=\cos(4t),\qquad s=\sin(4t)\end{gathered}}$
\par{\footnotesize The first two columns are the real and imaginary parts of $e^{(-3+4i)t}\vect{v}$.}
\vfill
\item The augmented matrix you would row reduce to satisfy the IC is
\par $M_{\mathrm{IC}}=\keybox{5.65in}{1.05in}{\left[\begin{array}{c|c}\Phi(0)&\vect{y}(0)\end{array}\right]
\quad\text{or}\quad\left[\begin{array}{rrr|r}1&0&0&6\\1&1&0&2\\1&1&1&9\end{array}\right]}$
\vfill
\item Not on the exam
\par $\vect{y}(t)=\keybox{5.65in}{1.25in}{e^{-3t}\begin{pmatrix}c&s\\c-s&c+s\\c-s&c+s\end{pmatrix}\begin{pmatrix}6\\-4\end{pmatrix}+7e^{5t}\vect{v}_3}$
\end{enumerate}
\end{enumerate}
